\vspace{1cm}
\textbf{Komponentliste:}
\begin{itemize}
\item S3: Hovedbryter (vippebryter, minimum 250 V DC)
\item S1: Ladeknapp (momentan, lavspent)
\item S2: Utladningsknapp (momentan, minimum 450 V DC, 0.5 A)
\item Q1: IRFP450 N-kanal MOSFET
\item D1a, D1b: To UF4007 i serie (hver med 10 M$\Omega$ parallelt)
\item D2: 15 V zenerdiode (BZX55C15)
\item D3: Grønn LED
\item D4: 1N4148 signaldiode (gatebeskyttelse)
\item D5: UF4007 (snubber-diode)
\item R1: 1 k$\Omega$ (gate-motstand)
\item R2: 10 k$\Omega$ (snubber-motstand)
\item R11: 1 k$\Omega$ / 25 W (utladningsmotstand, pulsratet)
\item R12a, R12b: To 110 k$\Omega$ / 1 W i serie (LED-strømbegrenser)
\item C1: 100 µF / 450 V elektrolytisk (puls-egnet)
\item C2: 10 nF / 500 V keramisk (snubber)
\item T1: Ferritttransformator (10:200:5 viklinger, E20/10/6 kjerne, N27, med luftgap)
\item G1: Trigget gnistgap (wolframelektroder, ~0.5 mm, med triggerelektrode)
\item L1: Flat spiralspole (4 vindinger, 10 AWG)
\item F1: 2 A sikring
\end{itemize}
