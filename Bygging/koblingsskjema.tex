\vspace{1cm}
\textbf{Komponentliste:}
\begin{itemize}
\item S3: Hovedbryter (50 V DC)
\item S1: Ladeknapp (momentan, lavspent)
\item S2: Utladningsknapp (momentan, min. 450 V DC, 0,5 A)
\item Q1: IRFP450 N-kanal MOSFET
\item D1a, D1b: UF4007 (to i serie, hver med 10 M$\Omega$ parallelt)
\item D2: 15 V zenerdiode (BZX55C15)
\item D3: Grønn LED
\item D4: 1N4148
\item D5: UF4007 (snubber)
\item R1: 1 k$\Omega$ / 1 W (gate)
\item R2: 10 k$\Omega$ / 0,25 W (snubber)
\item R11: 1 k$\Omega$ / 25 W (utladning, pulsratet)
\item R12a, R12b: 110 k$\Omega$ / 1 W (LED)
\item R\_bleed: 1 M$\Omega$ / 1 W (bleeder)
\item C1: 100 µF / 450 V (puls-egnet)
\item C2: 10 nF / 500 V (snubber)
\item T1: Ferritttransformator (E20/10/6, N27, luftgap, 10:200:5)
\item G1: Trigget gnistgap (wolfram, ~0,5 mm, triggerelektrode)
\item L1: Flat spiralspole (4 vindinger, 10 AWG)
\item F1: 2 A sikring
\end{itemize}
