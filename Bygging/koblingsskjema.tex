% Batteri 45V med sikring og hovedbryter

\draw (0,1) to[fuse, l=F1 (2A)] (0,2);
\draw (0,2) to[switch, l=S3] (2,2);
% Ladeknapp S1
\draw (2,2) to[switch, l=S1] (4,2);
% Primærvikling (10 t) med dot
\draw (4,2) to[inductor, l=10\,t, name=prim] (4,0.5);
\node at (3.5,1.25) {$\bullet$};
% MOSFET Q1
\draw (4,0.5) node[nmos, anchor=drain] (Q1) {};
\draw (Q1.source) to (0,0);
% RCD-snubber: diode anode til drain, katode til R2||C2, andre ende til forsyning
\draw (4,0.5) to[diode, l=D5] (5.5,0.5);
\draw (5.5,0.5) -- (5.5,1) to[resistor, l=R2] (5.5,2) -- (4,2);
\draw (5.5,0.5) -- (6,0.5) to[capacitor, l=C2] (6,2) -- (4,2);
% Tilbakekoblingsvikling (5 t) med dot
\draw (4,2) -- (7,2) to[inductor, l=5\,t, name=fb] (7,0.5);
\node at (7.5,1.25) {$\bullet$};
% Kondensator C3 i serie med tilbakekobling
\draw (7,0.5) to[capacitor, l=C3] (7,-0.5);
% Gate-nettverk
\draw (7,-0.5) -- (Q1.gate);
\draw (Q1.gate) to[resistor, l=R3] (Q1.source);
% Gate-clamp: zener D2 (katode til gate), diode D4 (katode til gate, anode til zener anode)
\draw (Q1.gate) to[Zener diode, l=D2] ++(0,-1.5) to[diode, l=D4] (Q1.source);
% Startmotstand
\draw (4,2) to[resistor, l=R\_start] (Q1.gate);
% Sekundærvikling (200 t) med dot (motsatt polaritet)
\draw (0,0) to[inductor, l=200\,t, name=sec] (10,3);
\node at (5,3.5) {$\bullet$};
% Likeretterdioder (to i serie)
\draw (10,3) to[diode, l=D1a] (10,2) to[diode, l=D1b] (10,1);
% Hovedkondensator C1 (500 V)
\draw (10,1) to[capacitor, l=C1] (10,0) -- (0,0);
% Bleeder-motstand
\draw (10,1) -- (12,1) to[resistor, l=R\_bleed] (12,0) -- (0,0);
% Utladningskrets
\draw (10,1) -- (13,1) to[switch, l=S2] (13,0) to[resistor, l=R11] (13,-1) -- (0,-1) -- (0,0);
% LED-indikator
\draw (10,1) -- (11,1) to[resistor, l=110\,k$\Omega$] (11,0) to[resistor, l=110\,k$\Omega$] (11,-1) to[led, l=D3] (11,-2) -- (0,-2) -- (0,-1);
% Gnistgap og spole (serie)
\draw (10,1) -- (10,2) to[sparkgap, l=G1] (10,3);
\draw (10,3) -- (12,3) to[inductor, l=L1] (12,1) -- (10,1);
% Piezo-trigger (isolert)
\draw (10,2.5) -- (9,2.5) node[left] {Piezo};
% Jord
\draw (0,0) node[ground] {};
\end{circuitikz}

\vspace{1cm}
\textbf{Komponentliste:}
\begin{itemize}
\item S3: Hovedbryter (50 V DC)
\item S1: Ladeknapp (momentan, lavspent)
\item S2: Utladningsknapp (momentan, min. 450 V DC, 0,5 A)
\item Q1: IRFP450 N-kanal MOSFET
\item D1a, D1b: UF4007 (to i serie)
\item D2: 15 V zenerdiode (BZX55C15)
\item D3: Grønn LED
\item D4: 1N4148
\item D5: UF4007 (snubber)
\item R\_start: 100 k$\Omega$ / 0,25 W (oppstart)
\item R2: 10 k$\Omega$ / 0,25 W (snubber)
\item R3: 10 k$\Omega$ / 0,25 W (gate-utladning)
\item R11: 1 k$\Omega$ / 25 W (utladning, pulsratet)
\item R12a, R12b: 110 k$\Omega$ / 1 W (LED)
\item R\_bleed: 1 M$\Omega$ / 1 W (bleeder)
\item C1: 100 µF / 500 V (puls-egnet)
\item C2: 10 nF / 500 V (snubber)
\item C3: 100 nF / 100 V (DC-blokkering gate)
\item T1: Ferritttransformator (E20/10/6, N27, 0.1 mm luftgap, 10:200:5)
\item G1: Trigget gnistgap (wolfram, ~0,5 mm, triggerelektrode)
\item L1: Flat spiralspole (4 vindinger, 10 AWG)
\item F1: 2 A sikring
\end{itemize}
