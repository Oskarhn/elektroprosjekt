\textbf{Merk: Alle beregninger er teoretiske. Faktiske verdier avhenger av målte parametere.}

\section{Hovedkondensator og spole}

\subsection{Spesifikasjoner}
\begin{itemize}
\item Kondensator: $C = 100\,\mu\text{F}$, merkespenning $500\,\text{V}$ (lades til $V_0 = 450\,\text{V}$)
\item Spole: Flat spiral, $N = 4$ vindinger, indre diameter $20\,\text{mm}$, ytre diameter $100\,\text{mm}$, tråd: $10\,\text{AWG}$ ($2.588\,\text{mm}$ diameter)
\end{itemize}

\subsection{Induktans (Wheelers formel)}
Indre radius $r_i = 10\,\text{mm}$, ytre radius $r_o = 50\,\text{mm}$.
Middelradius $r = (r_i + r_o)/2 = 30\,\text{mm} = 0.03\,\text{m}$.
Viklingsbredde $w = r_o - r_i = 40\,\text{mm} = 0.04\,\text{m}$.
I tommer: $r = 1.1811''$, $w = 1.5748''$.
\begin{equation}
L = \frac{r^2 N^2}{8r + 11w} = \frac{1.1811^2 \cdot 4^2}{8 \cdot 1.1811 + 11 \cdot 1.5748} = \frac{1.395 \cdot 16}{9.4488 + 17.3228} = \frac{22.32}{26.7716} \approx 0.8336\,\mu\text{H}.
\end{equation}
Estimat: $L = 0.83\,\mu\text{H}$ (må måles).

\subsection{Energi}
\begin{equation}
E_0 = \frac{1}{2} C V_0^2 = 0.5 \cdot 100 \times 10^{-6} \cdot 450^2 = 10.125\,\text{J} \approx 10\,\text{J}.
\end{equation}

\section{RLC-utladning}

\subsection{Karakteristisk impedans}
\begin{equation}
Z_0 = \sqrt{\frac{L}{C}} = \sqrt{\frac{0.83 \times 10^{-6}}{100 \times 10^{-6}}} = \sqrt{0.0083} \approx 0.0911\,\Omega.
\end{equation}

\subsection{Ideell toppstrøm}
\begin{equation}
I_{\text{peak, ideal}} = \frac{V_0}{Z_0} = \frac{450}{0.0911} \approx 4940\,\text{A}.
\end{equation}

\subsection{Dempet analyse}
Antatt total motstand $R = 0.1\,\Omega$ (må måles).
\begin{align}
\alpha &= \frac{R}{2L} = \frac{0.1}{2 \cdot 0.83 \times 10^{-6}} \approx 60241\,\text{s}^{-1}, \\
\omega_0 &= \frac{1}{\sqrt{LC}} = \frac{1}{\sqrt{0.83 \times 10^{-6} \cdot 100 \times 10^{-6}}} \approx 109719\,\text{rad/s}, \\
\omega_d &= \sqrt{\omega_0^2 - \alpha^2} \approx 91700\,\text{rad/s}, \\
f_d &= \frac{\omega_d}{2\pi} \approx 14.6\,\text{kHz}.
\end{align}
Første topp ved $t_p$ der $\tan(\omega_d t_p) = \omega_d/\alpha$:
\begin{equation}
\omega_d t_p = \arctan(91700/60241) \approx 0.99\,\text{rad} \Rightarrow t_p \approx 10.8\,\mu\text{s}.
\end{equation}
Toppstrøm:
\begin{equation}
I_{\text{peak}} = \frac{V_0}{\omega_d L} e^{-\alpha t_p} \sin(\omega_d t_p) \approx 2580\,\text{A}.
\end{equation}
Periode: $T_d = 1/f_d \approx 68.5\,\mu\text{s}$. Envelope-tidskonstant: $\tau = 1/\alpha \approx 16.6\,\mu\text{s}$.

\section{Magnetfelt}
Vindingenes senterradier: $r_k = 15, 25, 35, 45\,\text{mm}$.
\begin{equation}
B(0) = \frac{\mu_0 I}{2} \sum_{k=1}^{4} \frac{1}{r_k} = \frac{4\pi\times10^{-7} \cdot 2580}{2} \left( \frac{1}{0.015} + \frac{1}{0.025} + \frac{1}{0.035} + \frac{1}{0.045} \right) \approx 0.31\,\text{T}.
\end{equation}

\section{Pickup-spenning}
Pickup-spole: $N_p = 5$, diameter $2\,\text{cm}$ $\rightarrow$ areal $A_p = 3.14\times10^{-4}\,\text{m}^2$, avstand $z = 0.2\,\text{m}$.
Feltet ved pickup: $B(z) \approx 0.86\,\text{mT}$ (beregnet i script).
Maksimal $di/dt$: $di/dt(0) = V_0/L = 450 / 0.83\times10^{-6} \approx 5.42\times10^8\,\text{A/s}$.
\begin{equation}
\frac{dB}{dt}(z) = \frac{\mu_0}{2} \sum_{k=1}^{4} \frac{r_k^2}{(r_k^2+z^2)^{3/2}} \cdot \frac{di}{dt}.
\end{equation}
\begin{equation}
\mathcal{E}_{\text{max}} = N_p A_p \frac{dB}{dt} \approx 5 \cdot 3.14\times10^{-4} \cdot 1.64\times10^5 \approx 0.258\,\text{V} = 258\,\text{mV}.
\end{equation}
Faktisk verdi avhenger av nøyaktig geometri og måles.

\section{Komponentpåkjenninger}

\subsection{Dioder}
Reversspenning over likeretter:
\begin{equation}
V_{\text{rev}} = V_{\text{ut}} + \frac{N_s}{N_p} V_{\text{inn}} = 450 + 20 \cdot 45 = 1350\,\text{V}.
\end{equation}
To UF4007 i serie gir $2000\,\text{V}$ rating, men transientdeling må verifiseres.

\subsection{MOSFET}
Drain-spenning: $V_{\text{ds}} = V_{\text{inn}} + V_{\text{reflektert}} + V_{\text{leak}}$. Reflektert: $V_{\text{ut}}/(N_s/N_p) = 450/20 = 22.5\,\text{V}$. Lekkinduktansspike må dempes av snubber. IRFP450 tåler $500\,\text{V}$, men margin må verifiseres ved måling.

\subsection{Snubber}
Med $L_p = 34.5\,\mu\text{H}$ (gapped), lekkinduktans $1.7\,\mu\text{H}$, toppstrøm $\sim 2\,\text{A}$: energi $0.5 \cdot 1.7\times10^{-6} \cdot 4 = 3.4\,\mu\text{J}$. Snubberverdier $10\,\text{nF}$, $10\,\text{k}\Omega$ gir demping, men endelig verifisering krever måling av drain-transient.

\section{Transformator}
E20/10/6 N27 med 0.1 mm luftgap: $A_L \approx 345\,\text{nH/t}^2$.
\begin{align}
L_p &= 345 \times 10^{-9} \cdot 10^2 = 34.5\,\mu\text{H}, \\
L_s &= 34.5\,\mu\text{H} \cdot (200/10)^2 = 13.8\,\text{mH}.
\end{align}

\section{Repetisjonsrate}
Ladetid avhenger av ladestrøm, som må måles. Med antatt gjennomsnittlig ladestrøm $30\,\text{mA}$: $t = C V / I = 0.045 / 0.03 = 1.5\,\text{s}$ (hypotetisk). Faktisk ladetid forventes $5$--$10\,\text{s}$ basert på erfaring med 9V-batterier.

\section{Konklusjon}
Alle tall er teoretiske. Fysiske målinger er nødvendige for validering.
